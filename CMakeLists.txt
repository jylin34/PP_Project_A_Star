cmake_minimum_required(VERSION 3.16)
project(ParallelAStarResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(astar_core src/graph.cpp src/search.cpp src/parallel.cpp)
target_include_directories(astar_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(astar_core PUBLIC Threads::Threads)
target_compile_options(astar_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(parallel_astar src/main.cpp)
target_link_libraries(parallel_astar PRIVATE astar_core)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_search tests/test_search.cpp)
    target_link_libraries(test_search PRIVATE astar_core)
    add_test(NAME core_correctness COMMAND test_search)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME parallel_stress COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parallel_stress.py $<TARGET_FILE:parallel_astar>)
        set_tests_properties(parallel_stress PROPERTIES TIMEOUT 90)
    endif()
endif()
